    <script>
        const { createApp, ref, onMounted, nextTick } = Vue;

        createApp({
            setup() {
                const startDate = ref('2026-06-15');
                const endDate = ref('2026-06-20');
                const itineraries = ref([
                    { time: '09:00', location: '那霸機場 (OKA)' },
                    { time: '11:30', location: '首里城' },
                    { time: '14:00', location: '國際通美食街' }
                ]);

                const addPlan = () => {
                    itineraries.value.push({ time: '12:00', location: '' });
                    nextTick(() => {
                        lucide.createIcons();
                    });
                };

                const removePlan = (index) => {
                    itineraries.value.splice(index, 1);
                };

                const openMap = (location) => {
                    if(!location) return alert('請先輸入地點名稱');
                    const url = `https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(location)}`;
                    window.open(url, '_blank');
                };

                onMounted(() => {
                    lucide.createIcons();
                });

                return {
                    startDate, endDate, itineraries,
                    addPlan, removePlan, openMap
                }
            }
        }).mount('#app');
    </script>
